# ================================================
#  外掛模擬器 設定檔 (config.txt)
#  修改下列參數，存檔後重新執行 npm start，觀察日誌與結算效率。
#  井字號 # 之後為註解。
# ================================================

# --- 練功地點 ---
map B               # 在哪張圖練功：A 新手草原 / B 哥布林森林 / C 半獸人谷地
saveMap A           # 存檔點（死亡後復活處），預設 A

# --- 行為模式 ---
attackMode 2        # 0 不攻擊 / 1 被動(被打才還手) / 2 主動攻擊
itemsTakeAuto 1     # 0 不撿 / 1 脫離戰鬥才撿 / 2 看到就撿
sellAuto 1          # 0 不賣戰利品 / 1 戰利品全部賣掉

# --- 自動使用物品 ---
# 格式：useSelf_item [物品名稱] { 條件 }
# 條件可用 hp/sp，搭配 < 或 >，數字可加 % 表示佔最大值百分比。
# 物品名稱留空 = 關閉該區塊。一個區塊只能一件物品。
useSelf_item 補血藥水 {
    hp < 50%
}

useSelf_item 補魔藥水 {
    sp < 20%
}

# --- 自動購買物品 ---
# 格式：buyAuto [物品名稱] { npc 地圖, minAmount 下限, maxAmount 上限, disabled 0/1 }
# 身上數量低於 minAmount 時，自動補貨到 maxAmount。disabled 1 可停用。
buyAuto 補血藥水 {
    npc A
    minAmount 5
    maxAmount 20
    disabled 0
}
